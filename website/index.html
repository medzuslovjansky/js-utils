Direct Browser ESM Quickstart

            

              You can import these packages directly in any browser with
              <script type="module"> without any bundler or
              build step:
            

          

          

            
<script type="module">
  import { transliterate, FlavorisationBCP47 } from 'https://esm.sh/@interslavic/translit';
  import { inflect } from 'https://esm.sh/@interslavic/morphology';
  import { formatTokens } from 'https://esm.sh/@interslavic/conllu';
  import { distance } from 'https://esm.sh/@interslavic/levenshtein';

  // 1. Transliterate to Cyrillic
  console.log(transliterate('Dobry den!', 'isv-Cyrl')); // Добры ден!

  // 2. Inflect paradigm to CoNLL-U
  const tokens = inflect('žena');
  console.log(formatTokens(tokens));

  // 3. Measure cognate distance
  const dist = distance('rijeka', 'hr', 'rěka');
  console.log('Similarity score:', Math.max(0, 1 - dist));
</script>
